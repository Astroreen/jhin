# T41 QA - deliberate failure demo
# Copies of .vscode/launch.json are mutated under /tmp; the same
# validator that passes on the real file must fail here and name the
# problem. Real files are untouched (asserted at the end).
#
## (a) mainClass pointed at a wrong class
$ jq '.configurations[0].mainClass = "com.astroreen.jhin.kernel.NoSuchMain"' .vscode/launch.json > /tmp/opencode/launch-badclass.json
$ bash /tmp/opencode/t41-validate.sh /tmp/opencode/launch-badclass.json
ok: /tmp/opencode/launch-badclass.json parses as JSON
ok: program exists: frontend/lib/main.dart
ok: program exists: frontend/lib/main.dart
FAIL: mainClass com.astroreen.jhin.kernel.NoSuchMain has no source file at kernel/src/main/java/com/astroreen/jhin/kernel/NoSuchMain.java
exit: 1
#
## (b) program pointed at a missing dart entrypoint
$ jq '.configurations[2].program = "frontend/lib/missing.dart"' .vscode/launch.json > /tmp/opencode/launch-badprogram.json
$ bash /tmp/opencode/t41-validate.sh /tmp/opencode/launch-badprogram.json
ok: /tmp/opencode/launch-badprogram.json parses as JSON
FAIL: program named by /tmp/opencode/launch-badprogram.json does not exist: frontend/lib/missing.dart
ok: program exists: frontend/lib/main.dart
ok: mainClass com.astroreen.jhin.kernel.KernelMain -> kernel/src/main/java/com/astroreen/jhin/kernel/KernelMain.java
exit: 1
#
## (c) broken JSON (truncated file)
$ head -c 200 .vscode/launch.json > /tmp/opencode/launch-broken.json
$ bash /tmp/opencode/t41-validate.sh /tmp/opencode/launch-broken.json
FAIL: /tmp/opencode/launch-broken.json is not valid JSON: jq: parse error: Unfinished string at EOF at line 9, column 9
exit: 1
#
## real file untouched and still valid
$ bash /tmp/opencode/t41-validate.sh .vscode/launch.json
ok: .vscode/launch.json parses as JSON
ok: program exists: frontend/lib/main.dart
ok: program exists: frontend/lib/main.dart
ok: mainClass com.astroreen.jhin.kernel.KernelMain -> kernel/src/main/java/com/astroreen/jhin/kernel/KernelMain.java
exit: 0
