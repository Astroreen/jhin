T208 offline strategy — the served fetch-handler excerpts (printable substitute
for the unavailable offline.png; nothing visual is fabricated).

The strategy contract of /sw.js (verbatim from the served body — sw.js.txt
holds the full script): cache name + precache set, then the fetch strategies.

var CACHE = "jhin-shell-v1";
var CACHE_PREFIX = "jhin-shell-";
var PRECACHE = ["/", "/ui/assets/app.js", "/ui/assets/app.css"];
var NEVER_CACHE = ["/_jhin/bootstrap", "/ui/screen/", "/ui/action/", "/ui/stream"];

function neverCache(path) {
  for (var i = 0; i < NEVER_CACHE.length; i++) {
    if (path.indexOf(NEVER_CACHE[i]) === 0) {
      return true;
    }
  }
  return false;
}

function cacheFirst(request) {
  return caches.match(request).then(function (cached) {
    if (cached) {
      return cached;
    }
    return fetch(request).then(function (response) {
      if (response && response.ok) {
        var copy = response.clone();
        caches.open(CACHE).then(function (cache) {
          cache.put(request, copy);
        });
      }
      return response;
    });
  });
}

function networkFirst(request) {
  return fetch(request).then(function (response) {
    if (response && response.ok) {
      var copy = response.clone();
      caches.open(CACHE).then(function (cache) {
        cache.put(request, copy);
      });
    }
    return response;
  }).catch(function () {
    return caches.match(request);
  });
}

self.addEventListener("fetch", function (event) {
  var request = event.request;
  if (request.method !== "GET") {
    return;
  }
  var url = new URL(request.url);
  if (url.origin !== self.location.origin) {
    return;
  }
  if (neverCache(url.pathname)) {
    event.respondWith(fetch(request));
    return;
  }
  if (url.pathname.indexOf("/ui/assets/") === 0) {
    event.respondWith(cacheFirst(request));
    return;
  }
  if (url.pathname === "/") {
    event.respondWith(networkFirst(request));
    return;
  }
});

=== the offline render path, cross-referenced ===
On an offline navigation the networkFirst branch falls back to
caches.match(request) — the cached shell document (precache entry "/"). The
shell carries the paint target and the runtime does the rest:
  served shell: <div id="jhin-root"></div>
  client runtime: app.js renderOffline() (module 5 of the T204 client runtime)
    paints bootData.fallbacks.offline into #jhin-root when the bootstrap fetch
    fails — the art is server-authored in the bootstrap payload, contract-tested
    in unit scope (HtmlBootstrapRouteTest.fallbacksAreHtmlStrings,
    BootstrapContractTest.screensAndFallbacksAreHtmlFragments: fallbacks.offline
    is an HTML fragment carrying data-jhin-type=), and its paint target is
    asserted by WebShellRouteTest.carriesTheClientRootInTheProductionBranch
    ("the offline art of module 5 (renderOffline) paints into this root").
What stays unproven here is only the visual result in a live engine
(browser-unavailable.txt).
